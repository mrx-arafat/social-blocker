```js
// Schedule → guard-mode resolution. Pure, Date passed in for testability.
const HM = /^([01]?\d|2[0-3]):([0-5]\d)$/;

function toMin(hm) {
  const m = HM.exec(hm || "");
  return m ? Number(m[1]) * 60 + Number(m[2]) : null;
}

function windowValid(w) {
  return w && Array.isArray(w.days) && w.days.length > 0 &&
    toMin(w.start) != null && toMin(w.end) != null &&
    ["normal", "strict", "off"].includes(w.mode);
}

// Does window w cover `now`? Handles start>end as a span crossing midnight.
function covers(w, day, minutes) {
  const s = toMin(w.start), e = toMin(w.end);
  if (s <= e) return w.days.includes(day) && minutes >= s && minutes < e;
  // crosses midnight: tonight's leg, or the tail of yesterday's window
  if (w.days.includes(day) && minutes >= s) return true;
  const prev = (day + 6) % 7;
  return w.days.includes(prev) && minutes < e;
}

// settings needs: schedule {enabled, defaultMode, windows[]}, strictUntil.
export function resolveMode(settings, now = new Date()) {
  if ((settings.strictUntil || 0) > now.getTime()) return "strict";
  const sch = settings.schedule || {};
  if (!sch.enabled) return "normal";
  const day = now.getDay();
  const minutes = now.getHours() * 60 + now.getMinutes();
  for (const w of sch.windows || []) {
    if (windowValid(w) && covers(w, day, minutes)) return w.mode;
  }
  return sch.defaultMode === "off" ? "off" : "normal";
}

// Is any strict source currently active? (used by the options phrase gate)
export function strictActive(settings, now = new Date()) {
  return resolveMode(settings, now) === "strict";
}
```

```js
// Calm-day streaks + session-length estimates. Pure functions over stats.

// A day is "calm" when no daily limit wall was hit and total guarded time
// stayed under the global budget (budget 0 = budget check off).
export function dayQualifies(day, { budgetMinutes, sites }) {
  if (!day) return true; // no record = zero opens = calm
  for (const s of sites || []) {
    if (s.openLimit > 0 && (day.opens?.[s.domain] || 0) >= s.openLimit) return false;
    if (s.timeLimitMin > 0 && (day.time?.[s.domain] || 0) / 60 >= s.timeLimitMin) return false;
  }
  if (budgetMinutes > 0) {
    const totalSec = Object.values(day.time || {}).reduce((a, b) => a + b, 0);
    if (totalSec / 60 >= budgetMinutes) return false;
  }
  return true;
}

function nextDate(key) {
  const d = new Date(key + "T12:00:00"); // noon avoids DST edges
  d.setDate(d.getDate() + 1);
  const m = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${d.getFullYear()}-${m}-${day}`;
}

// Process every day after lastProcessedDate up to (excluding) todayKey.
// Missing records count as calm. Idempotent: keyed by lastProcessedDate.
export function rollStreak(allStats, streak, settings, todayKey) {
  const out = { ...streak };
  if (!out.lastProcessedDate) {
    // First run: start counting from today; nothing historical to judge fairly.
    const y = nextDate.bind(null);
    out.lastProcessedDate = prevDate(todayKey);
  }
  let cur = out.lastProcessedDate;
  while (nextDate(cur) < todayKey || nextDate(cur) === prevDate(todayKey) === false && nextDate(cur) < todayKey) break; // replaced below
  for (let k = nextDate(out.lastProcessedDate); k < todayKey; k = nextDate(k)) {
    out.current = dayQualifies(allStats[k], settings) ? out.current + 1 : 0;
    out.best = Math.max(out.best, out.current);
    out.lastProcessedDate = k;
  }
  return out;
}

function prevDate(key) {
  const d = new Date(key + "T12:00:00");
  d.setDate(d.getDate() - 1);
  const m = String(d.getMonth() + 1).padStart(2, "0");
  const day = String(d.getDate()).padStart(2, "0");
  return `${d.getFullYear()}-${m}-${day}`;
}

// Median minutes-per-open for a domain across recorded days; fallback 15.
export function medianSessionMin(allStats, domain) {
  const samples = [];
  for (const day of Object.values(allStats || {})) {
    const opens = day.opens?.[domain] || 0;
    const sec = day.time?.[domain] || 0;
    if (opens > 0 && sec > 0) samples.push(sec / 60 / opens);
  }
  if (samples.length < 5) return 15;
  samples.sort((a, b) => a - b);
  const mid = Math.floor(samples.length / 2);
  const med = samples.length % 2 ? samples[mid] : (samples[mid - 1] + samples[mid]) / 2;
  return Math.round(med);
}
```

```js
// Persuasive, reactance-safe pause-screen lines. Factual cost framing only —
// the message informs, the choice stays with the user.

function fmtH(min) {
  const h = Math.floor(min / 60), m = Math.round(min % 60);
  return h ? `${h}h ${m ? m + "m" : ""}`.trim() : `${m}m`;
}
const nth = (n) => n + (["th", "st", "nd", "rd"][((n % 100) - 20) % 10] || ["th", "st", "nd", "rd"][n % 100] || "th");

// ctx: { domain, visitsToday, minutesToday, minutesWeek, streak,
//        reclaimedWeekMin, stepAwaysWeek, worstHour, nowHour }
export function pickMessage(ctx, rand = Math.random) {
  if (ctx.streak >= 2) {
    return `${ctx.streak}-day calm streak. Continuing may end today's.`;
  }
  if (ctx.worstHour != null && ctx.worstHour === ctx.nowHour) {
    const h12 = ((ctx.worstHour + 11) % 12) + 1;
    const ap = ctx.worstHour >= 12 ? "PM" : "AM";
    return `${h12} ${ap} is your weakest hour — you usually open ${ctx.domain} around now.`;
  }
  if (ctx.visitsToday >= 3) {
    return `${nth(ctx.visitsToday)} visit today — ${fmtH(ctx.minutesToday)} on ${ctx.domain} so far.`;
  }
  if (ctx.minutesWeek > 120) {
    return `This week: ${fmtH(ctx.minutesWeek)} on ${ctx.domain}. That's a movie and a walk.`;
  }
  if (ctx.stepAwaysWeek >= 3 && ctx.reclaimedWeekMin > 0) {
    return `You've stepped away ${ctx.stepAwaysWeek} times this week — about ${fmtH(ctx.reclaimedWeekMin)} reclaimed.`;
  }
  const generic = [
    "Was this open a decision, or a reflex?",
    "What were you doing 30 seconds ago?",
    "The feed will be the same in an hour.",
    "You can always come back on purpose."
  ];
  return generic[Math.floor(rand() * generic.length)];
}
```

```js
// declarativeNetRequest session-rule construction. Pure — background applies.

export function domainRegex(domain) {
  const esc = domain.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  return `^https?://([^/]*\\.)?${esc}(/.*)?$`;
}

// Returns the full desired session-rule set for current state.
// passes: {domain: expiryMs} — sites with a live pass get no rule.
export function buildRules(settings, mode, passes, extOrigin, now = Date.now()) {
  if (!settings.enabled || mode === "off") return [];
  const rules = [];
  let id = 1;
  for (const site of settings.sites || []) {
    const rid = id++; // ids stay aligned to site order even when skipped
    if (!site.enabled) continue;
    if ((passes[site.domain] || 0) > now) continue;
    rules.push({
      id: rid,
      priority: 1,
      action: {
        type: "redirect",
        redirect: {
          regexSubstitution: `${extOrigin}pause.html?domain=${encodeURIComponent(site.domain)}&target=\\0`
        }
      },
      condition: {
        regexFilter: domainRegex(site.domain),
        resourceTypes: ["main_frame"]
      }
    });
  }
  return rules;
}
```
